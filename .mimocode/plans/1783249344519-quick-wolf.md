# Certus — Platform Integration Feature Plan

## Context

Certus needs to become platform-agnostic, supporting any trading platform (MetaTrader, online brokers, etc.). This feature introduces:
- Plugin architecture for platform adapters
- File-based data import with real-time file watching
- MQ4 Expert Advisor for MetaTrader v4 integration
- Full trade lifecycle capture with deviation analysis

## 3. File-Based Data Import Service

### File Watcher Service

```csharp
// src/Certus.Infrastructure/Platform/FileImportService.cs
public class FileImportService : IFileImportService, IDisposable
{
    private readonly Dictionary<string, FileSystemWatcher> _watchers = new();
    private readonly IPlatformDataParser _parser;
    private readonly IPlatformConnectionRepository _connectionRepo;
    private readonly IPortfolioRepository _portfolioRepo;
    private readonly ITradeRepository _tradeRepo;
    private readonly IDomainEventDispatcher _eventDispatcher;
    
    public void StartWatching(Guid connectionId, string filePath)
    {
        var watcher = new FileSystemWatcher
        {
            Path = Path.GetDirectoryName(filePath),
            Filter = Path.GetFileName(filePath),
            NotifyFilter = NotifyFilters.LastWrite | NotifyFilters.Size,
            EnableRaisingEvents = true
        };
        
        watcher.Changed += OnFileChanged;
        watcher.Created += OnFileCreated;
        watcher.Error += OnWatcherError;
        
        _watchers[connectionId.ToString()] = watcher;
    }
    
    private async void OnFileChanged(object sender, FileSystemEventArgs e)
    {
        // Debounce: wait 100ms for file write to complete
        await Task.Delay(100);
        
        // Read with retry for file locking
        var data = await ReadFileWithRetryAsync(e.FullPath, maxRetries: 3);
        if (data != null)
        {
            await ProcessImportDataAsync(data);
        }
    }
    
    private async Task<string?> ReadFileWithRetryAsync(string path, int maxRetries)
    {
        for (int i = 0; i < maxRetries; i++)
        {
            try
            {
                using var stream = File.Open(path, FileMode.Open, FileAccess.Read, FileShare.ReadWrite);
                using var reader = new StreamReader(stream);
                return await reader.ReadToEndAsync();
            }
            catch (IOException) when (i < maxRetries - 1)
            {
                await Task.Delay(50 * (i + 1)); // Exponential backoff
            }
        }
        return null;
    }
}
```

```mql4
   EventKillTimer();

{

{
```

## 7. Application Layer

### Services

```csharp
// src/Certus.Application/Platform/IPlatformService.cs
public interface IPlatformService
{
    Task<PlatformConnection> ConnectAsync(string platformId, PlatformConfig config);
    Task DisconnectAsync(Guid connectionId);
    Task<IReadOnlyList<PlatformConnection>> GetConnectionsAsync();
    Task ImportPortfolioAsync(Guid connectionId, string externalPortfolioId);
    Task SyncStrategiesAsync(Guid connectionId, Guid portfolioId);
    Task ImportTradesAsync(Guid connectionId, Guid strategyId, DateTime from, DateTime to);
}

// src/Certus.Application/Platform/PlatformService.cs
public class PlatformService : IPlatformService
{
    private readonly IPlatformPluginLoader _pluginLoader;
    private readonly IPlatformConnectionRepository _connectionRepo;
    private readonly IPortfolioRepository _portfolioRepo;
    private readonly ITradeRepository _tradeRepo;
    private readonly IFileImportService _fileImportService;
    private readonly IDomainEventDispatcher _eventDispatcher;
    
    public async Task<PlatformConnection> ConnectAsync(string platformId, PlatformConfig config)
    {
        var plugin = _pluginLoader.GetPlugin(platformId);
        var adapter = plugin.CreateAdapter(config);
        
        var result = await adapter.ConnectAsync(config);
        
        var connection = new PlatformConnection(
            Guid.NewGuid(),
            platformId,
            adapter.PlatformName,
            PlatformConnectionStatus.Connected,
            DateTime.UtcNow,
            config
        );
        
        await _connectionRepo.AddAsync(connection);
        
        // Start file watching if config specifies file path
        if (config is FilePlatformConfig fileConfig)
        {
            _fileImportService.StartWatching(connection.Id, fileConfig.FilePath);
        }
        
        await _eventDispatcher.DispatchAsync(new PlatformConnected
        {
            ConnectionId = connection.Id,
            PlatformId = platformId
        });
        
        return connection;
    }
    
    public async Task ImportPortfolioAsync(Guid connectionId, string externalPortfolioId)
    {
        var connection = await _connectionRepo.GetByIdAsync(connectionId);
        var plugin = _pluginLoader.GetPlugin(connection.PlatformId);
        var adapter = plugin.CreateAdapter(connection.Config);
        
        var portfolios = await adapter.GetPortfoliosAsync();
        var platformPortfolio = portfolios.FirstOrDefault(p => p.ExternalId == externalPortfolioId);
        
        if (platformPortfolio == null)
            throw new InvalidOperationException($"Portfolio {externalPortfolioId} not found");
        
        // Create Certus portfolio from platform data
        var portfolio = Portfolio.Create(
            platformPortfolio.Name,
            $"Imported from {connection.PlatformName}",
            0, // Target return - to be configured
            0, // Target Sharpe - to be configured
            platformPortfolio.Balance
        );
        
        portfolio.SetPlatformReference(connection.Id, externalPortfolioId);
        
        await _portfolioRepo.AddAsync(portfolio);
        
        await _eventDispatcher.DispatchAsync(new PortfolioImported
        {
            PortfolioId = portfolio.Id,
            ConnectionId = connectionId,
            ExternalId = externalPortfolioId
        });
    }
}
```
